47def _private_package(name: str, directory: Path) -> str:

74def construct(
75    root: str | Path,
76    *,
77    rho_source: float,
78    rho_target: float,
79    construction_seed: int = 2026,
80    output_dir: str | Path,
81) -> AlignmentDataset:
82    """Materialize the existing bilateral partial-overlap construction.
83
84    root contains the raw G1/G2 files and Alignment.json. The constructor
85    partitions known matched identities into shared/source-only/target-only
86    groups and excludes pre-existing unmatched identities. Ratios are node
87    overlap requests, with finite-size rounding recorded in metadata. Existing
88    outputs are never replaced, including dangling symbolic links.
89    """
90    from date_261023.construct_bilateral_partial import (
91        construct_dataset, load_raw_dataset, write_constructed_dataset,
92    )
93
94    source = Path(root).expanduser().resolve()
95    destination = Path(output_dir).expanduser().absolute()
96    if destination.exists() or destination.is_symlink():
97        raise FileExistsError(f"Output already exists: {destination}")
98    args = Namespace(
99        raw_dir_name=".", g1_file="G1.edgelist", g2_file="G2.edgelist",
100        g1_nodes_file="G1.nodes", g2_nodes_file="G2.nodes",
101        alignment_file="Alignment.json",
102    )
103    raw = load_raw_dataset(source.parent, source.name, args)
104    built = construct_dataset(
105        raw, rho_source, rho_target, construction_seed,
106        raw_dir_relative=str(source),
107    )
108    write_constructed_dataset(destination, built, force=False)
109    return load_dataset(destination)

112def register_algorithm(name: str, factory: Callable[..., Any]) -> None:
113    """Register a new name without replacing a built-in or existing adapter.
114
115    factory(parameters=..., seed=..., device=..., dataset_family=...) returns
116    an object with fit(graph1, graph2) -> AlignmentResult. The factory never
117    receives a Dataset. Custom code must preserve the unsupervised contract.
118    """
119    if not isinstance(name, str) or not name or name.strip() != name:
120        raise ValueError("Algorithm name must be a nonempty, trimmed string")
121    if name in _BUILTINS or name == "opgm" or name in _CUSTOM:
122        raise ValueError(f"Algorithm name is already reserved: {name}")
123    if not callable(factory):
124        raise TypeError("factory must be callable")
125    _CUSTOM[name] = factory

128def _topology(graph: nx.Graph) -> nx.Graph:
129    if not isinstance(graph, nx.Graph) or graph.is_directed() or graph.is_multigraph():
130        raise TypeError("align requires simple undirected NetworkX graphs")
131    if not len(graph) or set(graph) != set(range(len(graph))):
132        raise ValueError("Use nonempty graphs with contiguous local IDs 0..n-1; Dataset.load performs reindexing")
133    if nx.number_of_selfloops(graph):
134        raise ValueError("align requires graphs without self loops")
135    # Discard graph, node and edge attributes, including any caller annotations.
136    clean = nx.Graph()
137    clean.add_nodes_from(range(len(graph)))
138    clean.add_edges_from(graph.edges())
139    # Match the existing graph packet: seeded random walks can depend on
140    # adjacency iteration order even when the mathematical graph is identical.
141    for node in clean:
142        clean._adj[node] = {neighbor: clean._adj[node][neighbor] for neighbor in graph.neighbors(node)}
143    return clean

146class Algorithm:
147    """Configure an adapter, then call align(graph1, graph2).
148
149    The default device is CPU. Built-in adapter/config defaults apply; this
150    does not silently import a frozen campaign's family-specific parameters.
151    Registry construction is lazy and exceptions propagate to the caller.
152    """
153
154    def __init__(self, method: str, parameters: Mapping[str, Any] | None = None,
155                 *, seed: int = 2026, device: str = "cpu", dataset_family: str = ""):
156        if method == "opgm":
157            raise ValueError("Supervised OPGM-rs is protocol-incompatible; use the explicitly changed opgm_unsupervised adapter")
158        if method not in _BUILTINS and method not in _CUSTOM:
159            raise ValueError(f"Unknown algorithm {method!r}; built-ins: {', '.join(sorted(_BUILTINS))}")
160        if method == "alpine" and device not in {"cpu", "auto"}:
161            raise ValueError("The original-author ALPINE adapter supports CPU only; use device='cpu' or 'auto'")
162        self.method, self.parameters = method, copy.deepcopy(dict(parameters or {}))
163        forbidden = {"ground_truth", "gt", "true_k", "true_K", "true_overlap", "dataset", "data_root"}
164        if forbidden.intersection(self.parameters):
165            raise ValueError("Ground truth, true cardinality/overlap and dataset inputs are forbidden parameters")
166        self.seed, self.device, self.dataset_family = seed, device, dataset_family
167        self.config: dict[str, Any] | None = None
168
169    def align(self, graph1: nx.Graph, graph2: nx.Graph) -> AlignmentResult:
170        graphs = (_topology(graph1), _topology(graph2))
171        input_sizes = tuple(map(len, graphs))
172        random.seed(self.seed)
173        np.random.seed(self.seed)
174        parameters = copy.deepcopy(self.parameters)
175        if self.method in _CUSTOM:
176            model = _CUSTOM[self.method](parameters=parameters, seed=self.seed,
177                                         device=self.device, dataset_family=self.dataset_family)
178            self.config = parameters
179        else:
180            # Importing a built-in registry may require the project's model
181            # environment; custom adapters do not acquire that dependency.
182            import torch
183            torch.manual_seed(self.seed)
184            if self.method == "opgm_unsupervised":
185                from baseline.opgm_unsupervised.model import construct as make_model
186                model, self.config = make_model(parameters, self.seed, self.device)
187            elif self.method == "alpine":
188                from baseline.alpine_author_cpu import AuthorCPUAlpine
189                defaults = dict(mu=1.0, niter=10, inner_iterations=10, weight=2,
190                                sinkhorn_reg=1.0, sinkhorn_iterations=500,
191                                sinkhorn_tolerance=1e-5, dtype="float64", device="cpu")
192                unsupported = parameters.keys() - defaults.keys()
193                if unsupported:
194                    raise TypeError(f"Unsupported original-author ALPINE parameters: {', '.join(sorted(unsupported))}")
195                if parameters.get("device", "cpu") not in {"cpu", "auto"}:
196                    raise ValueError("The original-author ALPINE adapter supports CPU only")
197                self.config = dict(defaults, **parameters)
198                self.config.update(device="cpu", implementation="original_author_CPU_261030")
199                if self.config["niter"] != 10:
200                    raise ValueError("The original ALPINE code fixes ten outer iterations; niter must be 10")
201                if self.config["mu"] < 0 or self.config["weight"] not in {1, 2}:
202                    raise ValueError("ALPINE requires nonnegative mu and weight in {1, 2}")
203                model = AuthorCPUAlpine(self.config)
204            else:
205                from baseline.registry import construct as make_model
206                model, self.config = make_model(self.method, parameters, self.seed, self.device, self.dataset_family)
207        if not callable(getattr(model, "fit", None)):
208            raise TypeError("Algorithm factory must return an adapter with fit(graph1, graph2)")
209        context = contextlib.nullcontext()
210        if self.method == "s_gwl":
211            from baseline.sgwl_numerics_v2 import stable_proximal_state
212            context = stable_proximal_state()
213        with context as numerical:
214            result = model.fit(*graphs)
215        if numerical is not None:
216            result.metadata["numerical_revision"] = "retained_log_proximal_state_and_dual_gauge_v2"
217            result.metadata["numerical_revision_diagnostics"] = dict(numerical)
218        _validate_result(result, *input_sizes)
219        return result

222def _validate_result(result: AlignmentResult, n1: int, n2: int) -> dict[str, Any]:
223    if not isinstance(result, AlignmentResult):
224        raise TypeError("Expected bpupga.AlignmentResult")
225    result.validate_shapes(n1, n2)
226    validity = _metrics().validate_predicted_pairs(result.predicted_pairs, n1, n2)
227    if validity["out_of_range_count"]:
228        raise ValueError("Result contains out-of-range predicted pairs")
229    if result.metadata.get("nonfinite_guard_triggered"):
230        raise FloatingPointError("Adapter reported a nonfinite guard; no successful score is returned")
231    for field in ("score_matrix", "cost_matrix", "source_embeddings", "target_embeddings"):
232        value = getattr(result, field)
233        if value is not None and not np.isfinite(value).all():
234            raise FloatingPointError(f"Non-finite {field}")
235    return validity

238def _validate_ks(ks: Sequence[int]) -> tuple[int, ...]:
239    values = tuple(ks)
240    if not values or any(not isinstance(k, int) or isinstance(k, bool) or k < 1 for k in values):
241        raise ValueError("ks must contain positive integers")
242    return values

253    metadata = dataset.metadata

261def evaluate(result: AlignmentResult, dataset: AlignmentDataset,
262             *, ks: Sequence[int] = (1, 5, 10)) -> dict[str, Any]:
263    """Score native pairs and available same-fit rankings after inference.
264
265    Pair scoring removes exact duplicate pairs and reports endpoint conflicts.
266    Node classification additionally reports accuracy, balanced accuracy and
267    unmatched-node scores for complete references. No decoder or refit is run.
268    Rankings cover reference-matched queries and
269    do not establish rejection quality. GRAMPA uses the current soft transform
270    and also exposes its raw ranking; sparse candidates are not densified.
271    """
272    ks = _validate_ks(ks)
273    m = _metrics()
274    n1, n2 = len(dataset.graph1), len(dataset.graph2)
275    validity = _validate_result(result, n1, n2)
276    truth = dataset.ground_truth
277    reverse = {v: u for u, v in truth.items()}
278    report = {
279        "method": result.method, "output_variant": "native",
280        "pair": m.evaluate_pair_set(result.predicted_pairs, truth, n1, n2),
281        "matchability": m.evaluate_matchability(result.predicted_pairs, truth, n1, n2),
282        "overlap_estimation": m.evaluate_overlap_estimation(result.predicted_pairs, truth, n1, n2),
283        "validity": validity, "reference_note": _reference_note(dataset),
284        "runtime": {"native_reported_seconds": result.runtime_sec,
285                    "peak_memory_mb": result.peak_memory_mb},
286    }
287    report["node_classification"] = None
288    if _reference_limitation(dataset):
289        report["matchability"] = None
290        report["overlap_estimation"] = None
291        report["node_classification"] = None
292        report["skipped_metrics"] = {
293            "matchability": "Incomplete reference cannot identify all truly unmatched nodes",
294            "overlap_estimation": "Incomplete reference does not establish true overlap size",
295            "node_classification": "Incomplete reference cannot define matched/unmatched confusion counts",
296        }
297    else:
298        runtime = _private_package("bpupga._native_runtime", _ROOT / "evaluation/runtime")
299        classification = importlib.import_module(runtime + ".evaluate").node_classification
300        report["node_classification"] = classification(result.predicted_pairs, truth, n1, n2)
301
302    def directions(forward, backward):
303        return {"source_to_target": forward, "target_to_source": backward,
304                "bidirectional": {key: (forward[key] + backward[key]) / 2
305                                   for key in (*[f"hits@{k}" for k in ks], "mrr")}}
306
307    quality = None
308    if result.score_matrix is not None:
309        quality = np.asarray(result.score_matrix)
310        if result.larger_is_better is False:
311            quality = -quality
312        representation = "score_matrix"
313    elif result.cost_matrix is not None:
314        quality = -np.asarray(result.cost_matrix)
315        representation = "negative_cost_matrix"
316    if quality is not None:
317        def matrix_ranking(matrix):
318            return directions(m.evaluate_ranking_from_matrix(matrix, truth, ks),
319                              m.evaluate_ranking_from_matrix(matrix.T, reverse, ks))
320        report["ranking"] = matrix_ranking(quality)
321        report["ranking_policy"] = {"representation": representation,
322                                    "queries": "reference-matched nodes only",
323                                    "tie_break": "lower local candidate ID"}
324        if result.method.lower() == "grampa":
325            package = _private_package("bpupga._native_runtime", _ROOT / "evaluation/runtime")
326            soft = importlib.import_module(package + ".grampa_soft_ranking")
327            report["raw_ranking"] = report["ranking"]
328            report["ranking"] = matrix_ranking(soft.sinkhorn_soft_scores(quality))
329            report["ranking_policy"].update(soft.POLICY)
330    elif result.ranking_candidates is not None:
331        candidates = result.ranking_candidates
332        report["ranking"] = directions(
333            m.evaluate_ranking_from_candidates(candidates, truth, ks),
334            m.evaluate_ranking_from_candidates(((v, u, s) for u, v, s in candidates), reverse, ks))
335        report["ranking_policy"] = {"representation": "sparse_native_candidates",
336                                    "missing_reference_pair": "zero reciprocal rank"}
337    elif result.source_embeddings is not None and result.target_embeddings is not None:
338        source, target = result.source_embeddings, result.target_embeddings
339        report["ranking"] = directions(
340            m.evaluate_ranking_from_embeddings(source, target, truth, ks),
341            m.evaluate_ranking_from_embeddings(target, source, reverse, ks))
342        report["ranking_policy"] = {"representation": "euclidean_embedding_distance"}
343    else:
344        report["ranking"] = None
345        report["ranking_policy"] = {"skipped_reason": "No same-fit scores, candidates or paired embeddings"}
346    return report

349def diagnose(result: AlignmentResult, dataset: AlignmentDataset,
350             *, ks: Sequence[int] = (1, 5, 10), example_limit: int = 10,
351             max_dense_entries: int = 25_000_000) -> dict[str, Any]:
352    """Run the existing evaluation-only diagnostic module without saving files.
353
354    Requires BPUPGA/baseline261023/evaluation from the research checkout.
355    Endpoint-mask oracle comparisons are explicitly labeled in the report;
356    true-cardinality oracle decoding remains disabled. No result is refitted.
357    """
358    ks = _validate_ks(ks)
359    if example_limit < 0 or max_dense_entries < 1:
360        raise ValueError("example_limit must be nonnegative and max_dense_entries positive")
361    n1, n2 = len(dataset.graph1), len(dataset.graph2)
362    validity = _validate_result(result, n1, n2)
363    if _reference_limitation(dataset):
364        return {
365            "schema_version": 1, "method": result.method,
366            "native_output": {"pair": _metrics().evaluate_pair_set(result.predicted_pairs, dataset.ground_truth, n1, n2),
367                              "validity": validity},
368            "score_space": {"available": False, "skipped_reason": "Incomplete ground-truth reference"},
369            "skipped_diagnostics": {
370                "private_node_error_decomposition": "Unannotated nodes are not known private nodes",
371                "matchability_signal": "Incomplete reference cannot label true matchability",
372                "oracle_decoding_ladder": "Reference overlap masks and cardinality are incomplete",
373            },
374            "reference_note": _reference_note(dataset),
375            "leakage_statement": "Post-inference reference accounting only; no refit or oracle decoding",
376        }
377    legacy = _ROOT.parent / "BPUPGA/baseline261023/evaluation"
378    if not (legacy / "diagnostics.py").is_file():
379        raise FileNotFoundError(f"Optional diagnostic source is absent: {legacy / 'diagnostics.py'}")
380    package = _private_package("bpupga._legacy_evaluation", legacy)
381    module = importlib.import_module(package + ".diagnostics")
382    # Legacy diagnostics interprets score_matrix as quality regardless of the
383    # direction flag. Normalize a shallow dataclass copy, never mutate result.
384    diagnostic_result = result
385    if result.score_matrix is not None and result.larger_is_better is False:
386        diagnostic_result = replace(result, score_matrix=-np.asarray(result.score_matrix), larger_is_better=True)
387    report = module.diagnose_alignment_result(
388        diagnostic_result, dataset, ks=ks, example_limit=example_limit,
389        max_dense_entries=max_dense_entries, include_oracle_k=False,
390    )
391    report["reference_note"] = _reference_note(dataset)
392    return report

395@dataclass(frozen=True)
396class RunLog:
397    """Read an existing attempt's current result.json or started.json snapshot.
398
399    Does not scan matrices, rerun inference, write, watch or update a campaign.
400    Failure/running scores are None, never successful zeroes.
401    """
402
403    attempt_dir: Path
404    _record: dict[str, Any]
405    _failure: dict[str, Any] | None = None
406
407    @classmethod
408    def open(cls, attempt_dir: str | Path) -> "RunLog":
409        root = Path(attempt_dir).expanduser().resolve()
410        result = root / "result.json"
411        if not result.is_file():
412            result = root / "started.json"
413        record = _read_object(result)
414        if not isinstance(record.get("status"), str):
415            raise ValueError(f"Missing string status in {result}")
416        failure = root / "fit_failure.json"
417        return cls(root, record, _read_object(failure) if failure.is_file() else None)
418
419    @property
420    def status(self) -> str:
421        return self._record["status"]
422
423    @property
424    def native_metrics(self) -> dict[str, Any] | None:
425        if self.status != "SUCCESS":
426            return None
427        return copy.deepcopy(self._record.get("variants", {}).get("native"))
428
429    @property
430    def runtime(self) -> dict[str, Any]:
431        runtime = copy.deepcopy(self._record.get("runtime", {}))
432        if "end_to_end_seconds" in self._record:
433            runtime.setdefault("end_to_end_seconds", self._record["end_to_end_seconds"])
434        return runtime
435
436    @property
437    def errors(self) -> list[dict[str, Any]]:
438        errors = []
439        if self._record.get("error"):
440            errors.append({key: self._record[key] for key in ("status", "error", "traceback") if key in self._record})
441        if self._failure:
442            errors.append(copy.deepcopy(self._failure))
443        return errors
444
445    def summary(self) -> dict[str, Any]:
446        return {"attempt_dir": str(self.attempt_dir), "status": self.status,
447                "task": copy.deepcopy(self._record.get("task", {})),
448                "native_metrics": self.native_metrics, "runtime": self.runtime,
449                "errors": self.errors}
450

452def _read_object(path: Path) -> dict[str, Any]:
453    with path.open(encoding="utf-8") as handle:
454        value = json.load(handle)
455    if not isinstance(value, dict):
456        raise ValueError(f"Expected a JSON object in {path}")
457    return value

@dataclass(frozen=True)
class AlignmentDataset:
    root: Path
    name: str
    graph1: nx.Graph
    graph2: nx.Graph
    ground_truth: Dict[int, int]
    g1_old_to_new: Dict[int, int]
    g1_new_to_old: Dict[int, int]
    g2_old_to_new: Dict[int, int]
    g2_new_to_old: Dict[int, int]
    metadata: Dict[str, object]

    @property
    def source_only_nodes(self) -> set[int]:
        return set(self.graph1.nodes()) - set(self.ground_truth.keys())

    @property
    def target_only_nodes(self) -> set[int]:
        return set(self.graph2.nodes()) - set(self.ground_truth.values())

    @property
    def source_overlap_ratio(self) -> float:
        n = self.graph1.number_of_nodes()
        return len(self.ground_truth) / n if n else 0.0

    @property
    def target_overlap_ratio(self) -> float:
        n = self.graph2.number_of_nodes()
        return len(self.ground_truth) / n if n else 0.0

    @property
    def node_jaccard_overlap(self) -> float:
        n1 = self.graph1.number_of_nodes()
        n2 = self.graph2.number_of_nodes()
        shared = len(self.ground_truth)
        union = n1 + n2 - shared
        return shared / union if union else 0.0

    def pairs_to_original(
        self, pairs: Iterable[Tuple[int, int]]
    ) -> list[tuple[int, int]]:
        return [
            (self.g1_new_to_old[int(u)], self.g2_new_to_old[int(v)])
            for u, v in pairs
        ]

    def pairs_to_reindexed(
        self, pairs: Iterable[Tuple[int, int]]
    ) -> list[tuple[int, int]]:
        converted = []
        for u, v in pairs:
            if int(u) not in self.g1_old_to_new:
                raise ValueError(f"Unknown original G1 node id in prediction: {u}")
            if int(v) not in self.g2_old_to_new:
                raise ValueError(f"Unknown original G2 node id in prediction: {v}")
            converted.append((self.g1_old_to_new[int(u)], self.g2_old_to_new[int(v)]))
        return converted

    def summary(self) -> Dict[str, object]:
        def graph_statistics(graph: nx.Graph) -> Dict[str, int | float]:
            node_count = graph.number_of_nodes()
            component_sizes = [
                len(component) for component in nx.connected_components(graph)
            ]
            largest = max(component_sizes, default=0)
            return {
                "components": len(component_sizes),
                "isolates": nx.number_of_isolates(graph),
                "largest_component_ratio": largest / node_count if node_count else 0.0,
                "average_degree": (
                    2.0 * graph.number_of_edges() / node_count if node_count else 0.0
                ),
                "density": nx.density(graph),
            }

        mapped_source_edges = {
            tuple(sorted((self.ground_truth[u], self.ground_truth[v])))
            for u, v in self.graph1.edges()
            if u in self.ground_truth and v in self.ground_truth
        }
        matched_targets = set(self.ground_truth.values())
        target_core_edges = {
            tuple(sorted((u, v)))
            for u, v in self.graph2.edges()
            if u in matched_targets and v in matched_targets
        }
        common_core_edges = mapped_source_edges & target_core_edges
        union_core_edges = mapped_source_edges | target_core_edges
        n1 = self.graph1.number_of_nodes()
        n2 = self.graph2.number_of_nodes()
        shared = len(self.ground_truth)
        return {
            "name": self.name,
            "root": str(self.root),
            "construction_protocol": self.metadata.get("protocol", self.metadata.get("construction")),
            "g1_nodes": n1,
            "g1_edges": self.graph1.number_of_edges(),
            "g2_nodes": n2,
            "g2_edges": self.graph2.number_of_edges(),
            "ground_truth_pairs": shared,
            "source_only_nodes": len(self.source_only_nodes),
            "target_only_nodes": len(self.target_only_nodes),
            "source_overlap_ratio": self.source_overlap_ratio,
            "target_overlap_ratio": self.target_overlap_ratio,
            "node_jaccard_overlap": self.node_jaccard_overlap,
            "source_private_ratio": len(self.source_only_nodes) / n1 if n1 else 0.0,
            "target_private_ratio": len(self.target_only_nodes) / n2 if n2 else 0.0,
            "size_ratio": min(n1, n2) / max(n1, n2) if max(n1, n2) else 0.0,
            "g1_structure": graph_statistics(self.graph1),
            "g2_structure": graph_statistics(self.graph2),
            "ground_truth_core": {
                "g1_edges": len(mapped_source_edges),
                "g2_edges": len(target_core_edges),
                "common_edges": len(common_core_edges),
                "edge_jaccard": (
                    len(common_core_edges) / len(union_core_edges)
                    if union_core_edges
                    else 0.0
                ),
            },
        }

@dataclass
class AlignmentResult:
    method: str
    predicted_pairs: List[Tuple[int, int]]
    runtime_sec: float
    peak_memory_mb: Optional[float] = None
    score_matrix: Optional[np.ndarray] = None
    cost_matrix: Optional[np.ndarray] = None
    source_embeddings: Optional[np.ndarray] = None
    target_embeddings: Optional[np.ndarray] = None
    # Sparse algorithm-native candidate scores.  Unlike ``score_matrix``, an
    # omitted pair means that the method never retrieved that candidate.
    ranking_candidates: Optional[List[Tuple[int, int, float]]] = None
    larger_is_better: Optional[bool] = None
    # Ephemeral model-native signals consumed by a method-aware partial decoder.
    # They are intentionally not serialized as JSON because they may be dense.
    partial_decoder_data: Optional[Dict[str, np.ndarray]] = None
    metadata: Dict[str, object] = field(default_factory=dict)
    # Optional read-only traces returned by the baseline261023 Alpine model.
    stage_score_matrices: Dict[str, np.ndarray] = field(default_factory=dict)
    stage_diagnostics_metadata: Dict[str, Dict[str, object]] = field(default_factory=dict)

    def validate_shapes(self, num_source_nodes: int, num_target_nodes: int) -> None:
        expected_matrix_shape = (num_source_nodes, num_target_nodes)
        if self.score_matrix is not None and self.score_matrix.shape != expected_matrix_shape:
            raise ValueError(
                f"score_matrix shape {self.score_matrix.shape} != {expected_matrix_shape}"
            )
        if self.cost_matrix is not None and self.cost_matrix.shape != expected_matrix_shape:
            raise ValueError(
                f"cost_matrix shape {self.cost_matrix.shape} != {expected_matrix_shape}"
            )
        if self.source_embeddings is not None and self.source_embeddings.shape[0] != num_source_nodes:
            raise ValueError("source_embeddings row count does not match G1")
        if self.target_embeddings is not None and self.target_embeddings.shape[0] != num_target_nodes:
            raise ValueError("target_embeddings row count does not match G2")
        for stage_name, matrix in self.stage_score_matrices.items():
            if matrix.shape != expected_matrix_shape:
                raise ValueError(
                    f"stage score matrix {stage_name!r} has shape "
                    f"{matrix.shape} != {expected_matrix_shape}"
                )
        if self.ranking_candidates is not None:
            seen = set()
            for source, target, score in self.ranking_candidates:
                pair = (int(source), int(target))
                if not (0 <= pair[0] < num_source_nodes and 0 <= pair[1] < num_target_nodes):
                    raise ValueError("ranking candidate contains an out-of-range node")
                if pair in seen:
                    raise ValueError("ranking_candidates contains a duplicate pair")
                if not np.isfinite(float(score)):
                    raise ValueError("ranking candidate score must be finite")
                seen.add(pair)
